from lifelines import KaplanMeierFitter

rng = np.random.default_rng(12345)

def bootstrapMeanStd(statFunc, sample, Nsamples = 200):
    
    if len(sample) < 10:
        return np.nan
    else:
        resampleStats = []
        for i in range(Nsamples):
            resampleStats.append(statFunc(np.array([sample[j] for j in rng.integers(low=0,high=len(sample),size=len(sample))])))
        return (np.mean(resampleStats,axis=0),np.std(resampleStats,axis=0))
    
def cohortDataLongevitySteepness(data):
    
    kmf = KaplanMeierFitter()
    kmf.fit(data[:, 0], data[:, 1])
    qs = np.interp([0.75,0.5,0.25], np.flip(np.array(kmf.survival_function_).transpose())[0], np.flip(kmf.timeline))
    return (qs[1], qs[1]/(qs[2]-qs[0]))

def cohortDataLongevityICV(data):
    
    kmf = KaplanMeierFitter()
    kmf.fit(data[:, 0], data[:, 1])
    mean = np.dot(kmf.survival_function_.index[1:]-kmf.survival_function_.index[:-1],kmf.survival_function_.values[:-1])
    tsquared = 2*np.dot(
        np.array(
            (kmf.survival_function_.index[1:]+kmf.survival_function_.index[:-1])/2
        ) * kmf.survival_function_.values[:-1].transpose()[0],
        np.array(kmf.survival_function_.index[1:]-kmf.survival_function_.index[:-1])
        )
    return (mean[0], (mean/np.sqrt(tsquared - mean*mean))[0])


def cohortDataLongevityICVtrimmed(data, frac=0.9):

    trimmed = np.sort(data,axis=0)[-int(len(data)*frac):]

    kmf = KaplanMeierFitter()
    kmf.fit(trimmed[:, 0], trimmed[:, 1])
    mean = np.dot(kmf.survival_function_.index[1:]-kmf.survival_function_.index[:-1],kmf.survival_function_.values[:-1])
    tsquared = 2*np.dot(
        np.array(
            (kmf.survival_function_.index[1:]+kmf.survival_function_.index[:-1])/2
        ) * kmf.survival_function_.values[:-1].transpose()[0],
        np.array(kmf.survival_function_.index[1:]-kmf.survival_function_.index[:-1])
        )
    return (mean[0], (mean/np.sqrt(tsquared - mean*mean))[0])
